    class Shield:
    """Holds one immutable policy, its enforcement engine, and opens sessions."""

    def __init__(
        self,
        policy: Policy | None = None,
        *,
        approval: ApprovalHandler | None = None,
        audit: AuditLog | str | Path | None = None,
        detectors: Sequence[Detector] = (),
        detector_timeout: float = 2.0,
    ) -> None:
        self.policy = policy if policy is not None else load_policy("strict")
        self.audit = AuditLog(audit) if isinstance(audit, (str, Path)) else audit
        self.engine = EnforcementEngine(
            self.policy, audit=self.audit, detectors=detectors, detector_timeout=detector_timeout
        )
        self.approval = approval

    @classmethod
    def from_policy(
        cls,
        source: str | Path | Mapping[str, Any],
        *,
        approval: ApprovalHandler | None = None,
        audit: AuditLog | str | Path | None = None,
        detectors: Sequence[Detector] = (),
        detector_timeout: float = 2.0,
    ) -> Shield:
        """Build a shield from a built-in policy name, a YAML file, or a mapping."""
        return cls(
            load_policy(source),
            approval=approval,
            audit=audit,
            detectors=detectors,
            detector_timeout=detector_timeout,
        )

    def close(self) -> None:
        """Flush a final signed checkpoint and close the audit log, if one is open."""
        if self.audit is not None:
            self.audit.close()

    def session(self, *, user: str | None = None, task: str | None = None) -> SessionScope:
        """Start a session for one agent run.

        Use as ``with shield.session(user=...) as s:`` or ``async with``.
        """
        return SessionScope(Session(user=user, task=task))

    def system_prompt(self, text: str) -> str:
        """Append this session's canary instruction to a system prompt.

        Build the prompt inside ``with shield.session(...)`` so the marker belongs to the
        run it protects. ``check_output`` and the engine's argument check look for it.
        """
        session = current_session()
        if session is None:
            raise AgentShieldError("system_prompt() needs an active session")
        return f"{text}\n\n{canary_instruction(session.canary)}"

    def check_output(self, text: str) -> str:
        """Return ``text``, or raise ``CanaryLeakError`` if it leaks the session marker.

        Call this on the model's final answer. It is a check, not a filter: the caller
        decides what to do with the exception.
        """
        session = current_session()
        if session is not None and session.canary in text:
            raise CanaryLeakError("output contains the session's canary marker")
        return text

    def langchain(self) -> Any:
        """Return middleware that routes a LangChain / LangGraph agent's tool calls here.

        Pass it to ``create_agent(..., middleware=[shield.langchain()])``. Needs the
        ``[langchain]`` extra.
        """
        try:
            from agentshield.langchain import ShieldMiddleware
        except ImportError as exc:  # pragma: no cover - depends on the environment
            raise ImportError(
                "Shield.langchain() needs the LangChain extra: "
                'pip install "agentshield-x[langchain]"'
            ) from exc
        return ShieldMiddleware(self)

    def tool(self, name: str, fn: ToolFunction | None = None) -> Any:
        """Wrap one sync or async callable so every call passes through the engine.

        Use as a decorator (``@shield.tool("read_file")``) or directly
        (``send = shield.tool("send_email", send)``). The wrapped function is unchanged
        apart from the check before it and the ``record`` after it.
        """
        if fn is None:
            return lambda function: self.tool(name, function)

        if inspect.iscoroutinefunction(fn):

            @wraps(fn)
            async def async_wrapper(*args: Any, **kwargs: Any) -> Any:
                call = await self._authorize_async(
                    ToolCall(name, _bind_arguments(fn, args, kwargs))
                )
                try:
                    result = await fn(*args, **kwargs)
                except BaseException:
                    self.engine.record(call)
                    raise
                await self._record_async(call, result)
                return self._spotlight_result(call, result)

            return async_wrapper

        @wraps(fn)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            call = self._authorize(ToolCall(name, _bind_arguments(fn, args, kwargs)))
            try:
                result = fn(*args, **kwargs)
            except BaseException:
                self.engine.record(call)
                raise
            self.engine.record(call, result)
            return self._spotlight_result(call, result)

        return wrapper

    def _spotlight_result(self, call: ToolCall, result: Any, *, blocks: bool = False) -> Any:
        """Delimit text from an ``untrusted`` tool; leave other results alone.

        A string is delimited. With ``blocks=True`` (model-bound content, such as a LangChain
        ``ToolMessage``), so is each string or ``{"type": "text"}`` block in a list. A plain
        Python tool's list result is returned as is, since the caller may use it as data.
        """
        rule = self.policy.tools.get(call.name)
        if rule is None or Label.UNTRUSTED not in rule.labels:
            return result
        if isinstance(result, str):
            return delimit(result, source=call.name)
        if blocks and isinstance(result, list):
            return [self._spotlight_block(call, block) for block in result]
        return result

    def _spotlight_block(self, call: ToolCall, block: Any) -> Any:
        if isinstance(block, str):
            return delimit(block, source=call.name)
        if isinstance(block, Mapping) and block.get("type") == "text":
            return {**block, "text": delimit(str(block.get("text", "")), source=call.name)}
        return block

    async def _authorize_async(self, call: ToolCall) -> ToolCall:
        """``_authorize`` without blocking the event loop while detectors run."""
        if not self.engine.detectors:
            return self._authorize(call)
        # to_thread copies the context, so the current session goes with it.
        return await asyncio.to_thread(self._authorize, call)

    async def _record_async(self, call: ToolCall, result: Any) -> None:
        if not self.engine.detectors:
            self.engine.record(call, result)
        else:
            await asyncio.to_thread(self.engine.record, call, result)

    def _authorize(self, call: ToolCall) -> ToolCall:
        """Run the engine and any approval handler for one call, or raise."""
        decision = self.engine.check(call)
        if decision.verdict is Verdict.DENY:
            raise PolicyViolationError(decision.reason)
        if decision.verdict is Verdict.ASK:
            approved = self.approval is not None and self.approval(call, decision)
            if not self.engine.resolve_approval(call, approved):
                if self.approval is None:
                    raise ApprovalRequired(decision.reason)
                if approved:
                    raise ApprovalRequired(
                        f"approved, but the call could not be counted or audited: {decision.reason}"
                    )
                raise ApprovalRequired(f"approval denied: {decision.reason}")
        return call

    def load_policy(source: str | Path | Mapping[str, Any]) -> Policy:
    """Load a built-in policy by name, a YAML file, or an already-parsed mapping."""
    if isinstance(source, Mapping):
        origin, data = "<mapping>", copy.deepcopy(dict(source))
    elif isinstance(source, str) and source in BUILTIN_POLICIES:
        origin, data = f"built-in {source!r}", copy.deepcopy(dict(BUILTIN_POLICIES[source]))
    else:
        path = Path(source)
        if not path.is_file():
            raise PolicyError(
                f"{str(source)!r} is neither a built-in policy "
                f"({', '.join(BUILTIN_POLICIES)}) nor an existing file"
            )
        origin = str(path)
        data = _read_yaml(path)

    data = _apply_extends(data, origin)
    try:
        return Policy.model_validate(data)
    except ValidationError as exc:
        raise PolicyError(_describe(exc, origin)) from None

    class Policy(_Model):
    """A validated, immutable policy."""

    version: Literal[2]
    name: str | None = None
    mode: Literal["enforce", "monitor"] = "enforce"
    unlabelled_tools: Verdict = Verdict.DENY
    tools: dict[str, ToolRule] = {}
    flows: tuple[Flow, ...] = ()
    limits: Limits = Limits()
    mcp: McpRules = McpRules()

    @model_validator(mode="after")
    def _flows_are_consistent(self) -> Policy:
        names = [flow.name for flow in self.flows]
        duplicates = sorted({n for n in names if names.count(n) > 1})
        if duplicates:
            raise ValueError(f"flows: duplicate flow names {duplicates}")
        for i, flow in enumerate(self.flows):
            if flow.when.tool is not None and flow.when.tool not in self.tools:
                raise ValueError(
                    f"flows.{i}.when.tool: {flow.when.tool!r} is not declared under tools"
                )
        return self

    @property
    def digest(self) -> str:
        """SHA-256 of the effective policy; stable across processes and YAML formatting."""
        canonical = json.dumps(
            self.model_dump(mode="python"),
            sort_keys=True,
            separators=(",", ":"),
            default=lambda o: sorted(o) if isinstance(o, frozenset) else str(o),
        )
        return hashlib.sha256(canonical.encode()).hexdigest()

    def terminal_approval(call: ToolCall, decision: Decision) -> bool:
    """Ask on the terminal. Only an explicit ``y`` or ``yes`` allows the call."""
    print(f"\nAgentShield needs approval for {call.name!r}")
    print(f"  reason: {decision.reason}")
    print(f"  rule:   {decision.rule or '(none)'}")
    print(f"  args:   {dict(call.args)!r}")
    try:
        answer = input("Allow this call once? [y/N] ").strip().lower()
    except EOFError:
        return False
    return answer in {"y", "yes"}

    class AuditLog:
    """One append-only file, resumed from its tail when reopened."""

    def __init__(
        self,
        path: str | Path,
        *,
        key: Any | None = None,
        checkpoint_every: int = 1000,
    ) -> None:
        if checkpoint_every < 1:
            raise AuditError("checkpoint_every must be at least 1")
        self.path = Path(path)
        self._key = key
        self._fh: TextIO | None = None
        self._sequence = 0
        self._last_hash = GENESIS
        self._since_checkpoint = 0
        self._signed = key is not None
        self._checkpoint_every = checkpoint_every
        self._lock = threading.RLock()
        self._load()
        self._fh = self.path.open("a", encoding="utf-8", newline="\n")

    @property
    def signed(self) -> bool:
        return self._signed

    @property
    def sequence(self) -> int:
        """The sequence number the next record will receive."""
        return self._sequence

    def write(
        self,
        call: ToolCall,
        decision: Decision,
        session: Session | None = None,
        *,
        policy: str | None = None,
    ) -> None:
        """Append one decision. Redaction happens before the record is hashed."""
        with self._lock:
            if self._fh is None:
                raise AuditError("the audit log is closed")
            record: dict[str, Any] = {
                "type": "record",
                "seq": self._sequence,
                "ts": _now(),
                "session": session.id if session is not None else None,
                "user": redact(session.user) if session is not None else None,
                "task": redact(session.task) if session is not None else None,
                "policy": policy,
                "tool": redact(call.name),
                "args": redact(dict(call.args)),
                "verdict": decision.verdict.value,
                "rule": decision.rule,
                "reason": redact(decision.reason),
                "prev": self._last_hash,
            }
            record["hash"] = record_digest(record)
            self._append(record)
            self._sequence += 1
            self._last_hash = str(record["hash"])
            self._since_checkpoint += 1
            if self._signed and self._since_checkpoint >= self._checkpoint_every:
                self._checkpoint()

    def close(self) -> None:
        """Sign a final checkpoint and close the file. Safe to call twice."""
        with self._lock:
            if self._fh is None:
                return
            try:
                if self._signed and self._since_checkpoint > 0:
                    self._checkpoint()
            finally:
                self._fh.close()
                self._fh = None

    def __enter__(self) -> AuditLog:
        return self

    def __exit__(self, *exc_info: object) -> None:
        self.close()

    def _append(self, record: Mapping[str, Any]) -> None:
        assert self._fh is not None
        self._fh.write(_line(record) + "\n")
        self._fh.flush()

    def _checkpoint(self) -> None:
        assert self._key is not None
        seq = self._sequence - 1
        digest = self._last_hash
        self._append(
            {
                "type": "checkpoint",
                "seq": seq,
                "hash": digest,
                "ts": _now(),
                "signature": _sign(self._key, checkpoint_message(seq, digest)),
            }
        )
        self._since_checkpoint = 0

    def _load(self) -> None:
        if not self.path.exists() or self.path.stat().st_size == 0:
            self._write_header()
            return

        text = self.path.read_text(encoding="utf-8")
        if text and not text.endswith("\n"):
            # A crash left a partial final line. Drop it before appending again.
            text = text[: text.rfind("\n") + 1]
            self.path.write_text(text, encoding="utf-8", newline="\n")
        lines = text.splitlines()
        if not lines:
            self._write_header()
            return

        header = _parse_line(lines[0], 1)
        if header.get("type") != "header":
            raise AuditError(f"{self.path}: first line is not an audit header")
        if header.get("version") != FORMAT_VERSION:
            raise AuditError(f"{self.path}: unsupported format version {header.get('version')!r}")

        algorithm = header.get("algorithm")
        if algorithm == "ed25519":
            if self._key is None:
                raise AuditError(f"{self.path} is signed; pass the same key to append to it")
            if header.get("public_key") != _public_key_b64(self._key):
                raise AuditError(f"{self.path} was signed with a different key")
            self._signed = True
        elif algorithm == "none":
            if self._key is not None:
                raise AuditError(f"{self.path} is unsigned; use a new file for signed records")
            self._signed = False
        else:
            raise AuditError(f"{self.path}: unknown signing algorithm {algorithm!r}")

        interval = header.get("checkpoint_every")
        if not isinstance(interval, int) or interval < 1:
            raise AuditError(f"{self.path}: invalid checkpoint_every in the header")
        self._checkpoint_every = interval

        last_checkpoint = -1
        records = 0
        for line_no, line in enumerate(lines[1:], start=2):
            record = _parse_line(line, line_no)
            kind = record.get("type")
            if kind == "record":
                seq = record.get("seq")
                digest = record.get("hash")
                if not isinstance(seq, int) or not isinstance(digest, str):
                    raise AuditError(f"{self.path}:{line_no}: malformed record")
                records += 1
                self._sequence = seq + 1
                self._last_hash = digest
            elif kind == "checkpoint":
                last_checkpoint = line_no
            else:
                raise AuditError(f"{self.path}:{line_no}: unknown record type")

        self._since_checkpoint = 0
        if last_checkpoint < 0:
            self._since_checkpoint = records
        else:
            for line_no, line in enumerate(lines[1:], start=2):
                if line_no > last_checkpoint and (
                    _parse_line(line, line_no).get("type") == "record"
                ):
                    self._since_checkpoint += 1

        report = verify_audit(self.path)
        if not report.valid:
            raise AuditError(
                f"{self.path} is not a valid audit chain:\n  " + "\n  ".join(report.errors)
            )

    def _write_header(self) -> None:
        header: dict[str, Any] = {
            "type": "header",
            "version": FORMAT_VERSION,
            "algorithm": "ed25519" if self._key is not None else "none",
            "public_key": _public_key_b64(self._key) if self._key is not None else None,
            "checkpoint_every": self._checkpoint_every,
        }
        self.path.write_text(_line(header) + "\n", encoding="utf-8", newline="\n")

    def verify_audit(path: str | Path) -> AuditReport:
    """Recompute the chain and every checkpoint signature; never raises."""
    errors: list[str] = []
    records = 0
    checkpoints = 0
    signed = False
    hashes: dict[int, str] = {}
    try:
        lines = Path(path).read_text(encoding="utf-8").splitlines()
    except OSError as exc:
        return AuditReport(False, 0, 0, False, (f"cannot read {path}: {exc}",))

    if not lines:
        return AuditReport(False, 0, 0, False, ("the file is empty",))

    try:
        header = _parse_line(lines[0], 1)
    except AuditError as exc:
        return AuditReport(False, 0, 0, False, (str(exc),))
    if header.get("type") != "header":
        return AuditReport(False, 0, 0, False, ("the first line is not an audit header",))
    if header.get("version") != FORMAT_VERSION:
        return AuditReport(
            False, 0, 0, False, (f"unsupported format version {header.get('version')!r}",)
        )
    algorithm = header.get("algorithm")
    if algorithm not in {"ed25519", "none"}:
        return AuditReport(False, 0, 0, False, (f"unknown algorithm {algorithm!r}",))
    signed = algorithm == "ed25519"
    public_key = header.get("public_key")
    if signed and not isinstance(public_key, str):
        errors.append("the header has no public key")

    expected_seq = 0
    previous = GENESIS
    last_signed_seq = -1

    for line_no, line in enumerate(lines[1:], start=2):
        try:
            record = _parse_line(line, line_no)
        except AuditError as exc:
            errors.append(str(exc))
            break

        kind = record.get("type")
        if kind == "record":
            seq = record.get("seq")
            digest = record.get("hash")
            if not isinstance(seq, int):
                errors.append(f"line {line_no}: sequence number is not an integer")
            elif seq != expected_seq:
                errors.append(f"line {line_no}: expected seq {expected_seq}, got {seq}")
            if record.get("prev") != previous:
                errors.append(f"line {line_no}: previous hash does not match")

            if isinstance(digest, str):
                if record_digest(record) != digest:
                    errors.append(f"line {line_no}: hash does not match the record")
                elif isinstance(seq, int):
                    hashes[seq] = digest
                previous = digest
            else:
                errors.append(f"line {line_no}: record has no hash")
            if isinstance(seq, int):
                expected_seq = seq + 1
            records += 1
        elif kind == "checkpoint":
            checkpoints += 1
            if not signed:
                errors.append(f"line {line_no}: checkpoint in an unsigned log")
                continue
            seq = record.get("seq")
            digest = record.get("hash")
            signature = record.get("signature")
            if not isinstance(seq, int) or not isinstance(digest, str):
                errors.append(f"line {line_no}: malformed checkpoint")
                continue
            if hashes.get(seq) != digest:
                errors.append(f"line {line_no}: checkpoint does not match a record")
            if not isinstance(signature, str) or not isinstance(public_key, str):
                errors.append(f"line {line_no}: checkpoint has no signature")
                continue
            try:
                _verify(public_key, signature, checkpoint_message(seq, digest))
            except AuditError:
                errors.append(f"line {line_no}: checkpoint signature does not verify")
            else:
                last_signed_seq = max(last_signed_seq, seq)
        else:
            errors.append(f"line {line_no}: unknown record type {kind!r}")

    if signed and records and checkpoints == 0:
        errors.append("the signed log has no checkpoints")

    return AuditReport(
        valid=not errors,
        records=records,
        checkpoints=checkpoints,
        signed=signed,
        errors=tuple(errors),
        unsigned_tail=sum(1 for seq in hashes if seq > last_signed_seq) if signed else 0,
    )

    def load_key(path: str | Path) -> Any:
    """Load an unencrypted PKCS#8 Ed25519 private key from a PEM file."""
    try:
        from cryptography.hazmat.primitives import serialization
        from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
    except ImportError as exc:  # pragma: no cover - depends on the environment
        raise ImportError(
            'signed audit logs need the audit extra: pip install "agentshield-x[audit]"'
        ) from exc
    key = serialization.load_pem_private_key(Path(path).read_bytes(), password=None)
    if not isinstance(key, Ed25519PrivateKey):
        raise AuditError(f"{str(path)!r} is not an Ed25519 private key")
    return key

    class ChunkedClassifier:
    """Flag text if any overlapping chunk of it scores at or above ``threshold``."""

    def __init__(
        self,
        score: Scorer,
        *,
        threshold: float = 0.5,
        chunk_chars: int = 1500,
        overlap: int = 200,
        name: str = "classifier",
    ) -> None:
        if not 0 <= overlap < chunk_chars:
            raise ValueError("overlap must be at least 0 and smaller than chunk_chars")
        self.score = score
        self.threshold = threshold
        self.chunk_chars = chunk_chars
        self.overlap = overlap
        self.name = name

    def flags(self, text: str) -> bool:
        chunks = chunk(text, self.chunk_chars, self.overlap)
        return bool(chunks) and max(self.score(chunks)) >= self.threshold
